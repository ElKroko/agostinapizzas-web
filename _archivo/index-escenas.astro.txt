<script>
  const recorrido = document.getElementById("recorrido") as HTMLDivElement | null;
  const escenas = Array.from(document.querySelectorAll<HTMLElement>(".escena"));
  const ticks = Array.from(document.querySelectorAll<HTMLButtonElement>(".indice-fijo button"));
  const ctx = document.getElementById("contexto-fijo");
  const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  let indice = 0;
  let bloqueado = false;

  function esHorizontal() {
    return window.matchMedia("(min-width: 1000px)").matches;
  }

  function marcarActivo(idx: number) {
    ticks.forEach((b, i) => b.setAttribute("aria-current", String(i === idx)));
    if (ctx && escenas[idx]) ctx.textContent = escenas[idx].dataset.nombre ?? "";
  }

  function irA(id: string) {
    const destino = document.getElementById(id);
    if (!destino) return;
    destino.scrollIntoView({ behavior: reduce ? "auto" : "smooth", block: "start", inline: "start" });
  }

  // Navegación por "diapositiva": cada gesto o tecla avanza exactamente una
  // escena — más robusto que traducir deltaY a scrollLeft de forma continua,
  // que en varios navegadores/dispositivos terminaba sin mover nada visible.
  function avanzar(delta: number) {
    if (bloqueado) return;
    const siguiente = indice + delta;
    if (siguiente < 0 || siguiente >= escenas.length) return;
    bloqueado = true;
    indice = siguiente;
    marcarActivo(indice);
    irA(escenas[indice].id);
    window.setTimeout(() => {
      bloqueado = false;
    }, 650);
  }

  ticks.forEach((btn, i) => {
    btn.addEventListener("click", () => {
      indice = i;
      marcarActivo(indice);
      irA(btn.dataset.ir!);
    });
  });

  recorrido?.addEventListener(
    "wheel",
    (e) => {
      if (!esHorizontal()) return;
      if (Math.abs(e.deltaX) > Math.abs(e.deltaY)) return; // gesto horizontal nativo del trackpad: no interceptar
      if (Math.abs(e.deltaY) < 8) return; // ruido de rueda de alta resolución
      e.preventDefault();
      avanzar(e.deltaY > 0 ? 1 : -1);
    },
    { passive: false }
  );

  document.addEventListener("keydown", (e) => {
    if (!esHorizontal()) return;
    const activo = document.activeElement as HTMLElement | null;
    if (activo && (activo.tagName === "INPUT" || activo.tagName === "TEXTAREA" || activo.isContentEditable)) return;
    if (e.key === "ArrowRight" || e.key === "ArrowDown" || e.key === "PageDown") {
      e.preventDefault();
      avanzar(1);
    } else if (e.key === "ArrowLeft" || e.key === "ArrowUp" || e.key === "PageUp") {
      e.preventDefault();
      avanzar(-1);
    }
  });

  // Mantiene `indice` fiel a lo que realmente se ve, sin importar el método
  // de navegación (swipe táctil, gesto nativo de trackpad, tecla, tick).
  if ("IntersectionObserver" in window) {
    const io = new IntersectionObserver(
      (entries) => {
        entries.forEach((entry) => {
          if (!entry.isIntersecting) return;
          const idx = escenas.indexOf(entry.target as HTMLElement);
          if (idx === -1) return;
          indice = idx;
          marcarActivo(idx);
        });
      },
      { threshold: 0.55 }
    );
    escenas.forEach((s) => io.observe(s));
  }

  // Escena 1 — paralaje sutil del recorte de Giulianna según el cursor.
  if (!reduce) {
    const cutout = document.getElementById("giulianna-cutout");
    const e1 = document.getElementById("e1");
    if (cutout && e1) {
      e1.addEventListener("mousemove", (ev) => {
        const r = e1.getBoundingClientRect();
        const px = (ev.clientX - r.left) / r.width - 0.5;
        const py = (ev.clientY - r.top) / r.height - 0.5;
        (cutout as HTMLElement).style.transform = `translate(${px * 14}px, ${py * 10}px)`;
      });
      e1.addEventListener("mouseleave", () => {
        (cutout as HTMLElement).style.transform = "translate(0,0)";
      });
    }
  }

  // Escena 2 — una sola foto cambia de encuadre en hover/foco.
  if (!reduce) {
    const swap = document.getElementById("foto-swap");
    if (swap) {
      const img = swap.querySelector("img") as HTMLImageElement;
      const list = (img.getAttribute("data-cycle") || "").split(",").filter(Boolean);
      const original = list[0];
      function mostrar(src: string) {
        img.style.opacity = "0";
        setTimeout(() => {
          img.src = src;
          img.style.opacity = "1";
        }, 130);
      }
      swap.addEventListener("mouseenter", () => list[1] && mostrar(list[1]));
      swap.addEventListener("focus", () => list[1] && mostrar(list[1]));
      swap.addEventListener("mouseleave", () => mostrar(original));
      swap.addEventListener("blur", () => mostrar(original));
    }
  }

  // Escena 3 — paralaje diferencial de la foto respecto al titular, según scroll.
  if (!reduce) {
    const foto3 = document.getElementById("foto-parallax");
    const e3 = document.getElementById("e3");
    if (foto3 && e3 && recorrido) {
      recorrido.addEventListener(
        "scroll",
        () => {
          if (!esHorizontal()) return;
          const r = e3.getBoundingClientRect();
          const progreso = 1 - Math.min(1, Math.max(0, r.left / window.innerWidth));
          (foto3 as HTMLElement).style.transform = `translateX(${(progreso - 0.5) * -26}px)`;
        },
        { passive: true }
      );
    }
  }

  // Escena 4 — foco/hover en una comuna resalta su punto en el mapa.
  const lista = document.getElementById("comuna-lista");
  if (lista) {
    const botones = Array.from(lista.querySelectorAll<HTMLButtonElement>("button"));
    botones.forEach((btn) => {
      function activar() {
        botones.forEach((b) => b.setAttribute("data-activa", String(b === btn)));
        document.querySelectorAll('[id^="pin-"]').forEach((p) => {
          p.setAttribute("fill", "var(--ink-suave)");
          p.setAttribute("r", "5");
        });
        const pin = document.getElementById("pin-" + btn.dataset.comuna);
        if (pin) {
          pin.setAttribute("fill", "var(--color-rojo)");
          pin.setAttribute("r", "7");
        }
      }
      btn.addEventListener("mouseenter", activar);
      btn.addEventListener("focus", activar);
    });
  }

  marcarActivo(0);
</script>
